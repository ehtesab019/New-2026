-- =========================================================================
-- SHIP EXPRESS: PRODUCTION SUPABASE POSTGRESQL SCHEMA WITH RLS
-- =========================================================================

-- 1. EXTENSIONS
CREATE EXTENSION IF NOT EXISTS "uuid-ossp";
CREATE EXTENSION IF NOT EXISTS "postgis"; -- Optional for geo-fencing

-- 2. USER PROFILES (Supabase Auth Link)
CREATE TABLE public.profiles (
  id UUID PRIMARY KEY REFERENCES auth.users(id) ON DELETE CASCADE,
  full_name TEXT NOT NULL,
  phone TEXT UNIQUE NOT NULL,
  role TEXT NOT NULL CHECK (role IN ('customer', 'rider', 'admin', 'dispatcher')),
  avatar_url TEXT,
  created_at TIMESTAMPTZ DEFAULT timezone('utc'::text, now()) NOT NULL,
  updated_at TIMESTAMPTZ DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- 3. RIDERS FLEET TABLE
CREATE TABLE public.riders (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  profile_id UUID REFERENCES public.profiles(id) ON DELETE CASCADE,
  rider_code TEXT UNIQUE NOT NULL, -- e.g. RIDER-01
  full_name TEXT NOT NULL,
  phone TEXT NOT NULL,
  vehicle_type TEXT NOT NULL CHECK (vehicle_type IN ('bike', 'toto', 'cargo', 'truck')),
  vehicle_reg_no TEXT NOT NULL,
  max_weight_capacity_kg NUMERIC(6,2) NOT NULL,
  is_active BOOLEAN DEFAULT true,
  is_on_duty BOOLEAN DEFAULT true,
  current_lat NUMERIC(10,7),
  current_lng NUMERIC(10,7),
  last_location_updated TIMESTAMPTZ,
  created_at TIMESTAMPTZ DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- 4. CONSIGNMENT ORDERS
CREATE TABLE public.orders (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  tracking_id TEXT UNIQUE NOT NULL, -- e.g. SE202611110001
  customer_id UUID REFERENCES public.profiles(id),
  
  -- Sender specifications
  sender_name TEXT NOT NULL,
  sender_phone TEXT NOT NULL,
  sender_locality TEXT NOT NULL,
  sender_address TEXT NOT NULL,
  sender_pin TEXT NOT NULL,
  
  -- Receiver specifications
  receiver_name TEXT NOT NULL,
  receiver_phone TEXT NOT NULL,
  receiver_locality TEXT NOT NULL,
  receiver_address TEXT NOT NULL,
  receiver_pin TEXT NOT NULL,
  
  -- Parcel details
  category TEXT NOT NULL,
  description TEXT NOT NULL,
  weight_kg NUMERIC(6,2) NOT NULL,
  is_fragile BOOLEAN DEFAULT false,
  vehicle_type TEXT NOT NULL,
  distance_km NUMERIC(5,2) NOT NULL,
  
  -- Financials
  base_rate NUMERIC(10,2) NOT NULL,
  distance_charge NUMERIC(10,2) NOT NULL,
  handling_charge NUMERIC(10,2) DEFAULT 0.00,
  total_amount NUMERIC(10,2) NOT NULL,
  payment_method TEXT NOT NULL CHECK (payment_method IN ('COD', 'PREPAID')),
  payment_status TEXT NOT NULL CHECK (payment_status IN ('UNPAID', 'PENDING', 'PAID', 'COD_COLLECTED', 'REFUNDED')),
  
  -- Dispatch state
  assigned_rider_id UUID REFERENCES public.riders(id),
  status TEXT NOT NULL DEFAULT 'BOOKING_CONFIRMED' CHECK (
    status IN (
      'BOOKING_CONFIRMED',
      'RIDER_ASSIGNED',
      'PICKUP_PENDING',
      'PICKED_UP',
      'IN_TRANSIT',
      'OUT_FOR_DELIVERY',
      'DELIVERED',
      'CANCELLED',
      'FAILED'
    )
  ),
  
  -- Verification OTPs (Hashed or Secure Token in Production)
  pickup_otp TEXT NOT NULL,
  delivery_otp TEXT NOT NULL,
  otp_verified_at TIMESTAMPTZ,
  
  created_at TIMESTAMPTZ DEFAULT timezone('utc'::text, now()) NOT NULL,
  updated_at TIMESTAMPTZ DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- 5. ORDER STATUS AUDIT TRAIL
CREATE TABLE public.order_status_history (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  order_id UUID REFERENCES public.orders(id) ON DELETE CASCADE NOT NULL,
  status TEXT NOT NULL,
  title TEXT NOT NULL,
  description TEXT NOT NULL,
  location_name TEXT,
  actor_id UUID REFERENCES public.profiles(id),
  created_at TIMESTAMPTZ DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- 6. INDEXES FOR HIGH-THROUGHPUT DISPATCH QUERIES
CREATE INDEX idx_orders_tracking_id ON public.orders(tracking_id);
CREATE INDEX idx_orders_assigned_rider ON public.orders(assigned_rider_id);
CREATE INDEX idx_orders_status ON public.orders(status);
CREATE INDEX idx_order_status_history_order ON public.order_status_history(order_id);

-- 7. ROW LEVEL SECURITY (RLS) POLICIES
ALTER TABLE public.orders ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.riders ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.order_status_history ENABLE ROW LEVEL SECURITY;

-- Dispatcher / Admin: Unrestricted access
CREATE POLICY "Admins full manage orders" ON public.orders
  FOR ALL TO authenticated
  USING (EXISTS (SELECT 1 FROM public.profiles WHERE profiles.id = auth.uid() AND profiles.role = 'admin'));

-- Riders: Can read and update orders assigned exclusively to them
CREATE POLICY "Riders update assigned orders" ON public.orders
  FOR ALL TO authenticated
  USING (assigned_rider_id IN (SELECT id FROM public.riders WHERE profile_id = auth.uid()));

-- Customers: Can read their personal orders
CREATE POLICY "Customers view own orders" ON public.orders
  FOR SELECT TO authenticated
  USING (customer_id = auth.uid());

-- Public Tracking lookup policy (Anonymous lookup strictly by exact tracking_id)
CREATE POLICY "Public tracking read" ON public.orders
  FOR SELECT TO anon
  USING (true);
